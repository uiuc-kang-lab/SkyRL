from collections import deque

def minimal_steps(s, t):
    target_a = s
    target_b = t

    # Initialize the queue with the initial state (empty, empty) and 0 steps
    queue = deque()
    queue.append( ( ('', ''), 0 ) )
    # Dictionary to store the minimal steps for each state
    visited = dict()
    visited[ ('', '') ] = 0

    while queue:
        a, b, steps = queue.popleft()

        if a == target_a and b == target_b:
            return steps

        # Generate all possible next states
        # Append to a
        new_a = a + 'a'
        new_b = b
        if (new_a, new_b) not in visited or steps + 1 < visited.get( (new_a, new_b), float('inf') ):
            visited[ (new_a, new_b) ] = steps + 1
            queue.append( (new_a, new_b, steps + 1) )

        # Append to b
        new_a = a
        new_b = b + 'a'
        if (new_a, new_b) not in visited or steps + 1 < visited.get( (new_a, new_b), float('inf') ):
            visited[ (new_a, new_b) ] = steps + 1
            queue.append( (new_a, new_b, steps + 1) )

        # Copy a to b
        new_a = a
        new_b = a + b
        if (new_a, new_b) not in visited or steps + 1 < visited.get( (new_a, new_b), float('inf') ):
            visited[ (new_a, new_b) ] = steps + 1
            queue.append( (new_a, new_b, steps + 1) )

        # Copy b to a
